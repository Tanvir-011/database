-----Explicit Cursor (এক্সপ্লিসিট কার্সর)

SET SERVEROUTPUT ON;

DECLARE
    -- ১. DECLARE: কার্সর তৈরি করা
    CURSOR c_emp IS
        SELECT ID, Name, Salary
        FROM Employees
        WHERE Department = 'IT';
 
    -- ডাটা রাখার জন্য ভ্যারিয়েবল
    v_id Employees.ID%TYPE;
    v_name Employees.Name%TYPE;
    v_salary Employees.Salary%TYPE;
BEGIN
    -- ২. OPEN: কার্সর ওপেন করা
    OPEN c_emp;
 
    LOOP
        -- ৩. FETCH: এক এক করে রো ভ্যারিয়েবলে নেওয়া
        FETCH c_emp INTO v_id, v_name, v_salary;
 
        -- ডাটা না পেলে লুপ থেকে বের হয়ে যাওয়া
        EXIT WHEN c_emp%NOTFOUND;
 
        -- ডাটা প্রিন্ট করা
        DBMS_OUTPUT.PUT_LINE('ID: ' || v_id || ' | Name: ' || v_name || ' | Salary: ' || v_salary);
    END LOOP;
 
    -- ৪. CLOSE: কার্সর ক্লোজ করা
    CLOSE c_emp;
END;
/





-----FOR LOOP সহ Explicit Cursor (সহজ ও আধুনিক পদ্ধতি)
SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_emp IS
        SELECT ID, Name, Salary FROM Employees;
BEGIN
    -- FOR Loop নিজেই OPEN, FETCH এবং CLOSE সামলে নেয়
    FOR emp_record IN c_emp LOOP
        DBMS_OUTPUT.PUT_LINE('Name: ' || emp_record.Name || ' - Salary: ' || emp_record.Salary);
    END LOOP;
END;
